      import: async (texts) => {
        const results = []
        for (const text of Array.isArray(texts) ? texts : [texts]) {
          const { entries, skipped } = importedAccountsIn(String(text ?? ""))
          for (const skip of skipped) results.push({ type: "failed", error: skip.why })
          for (const entry of entries) {
            const got = await prepareImportedAccount(entry, { renewOnce, fetcher })
            results.push(got.error ? { type: "failed", error: got.error } : { ...codeArtsRecordBody(got.cred), type: "success" })
          }
        }
        return results
      },
